import face_recognition
import cv2
import numpy as np
import os
import threading
import time

class SistemaAcessoUlife:
    def __init__(self):
        # Caminho absoluto tratado
        self.diretorio_fotos = r"C:\Users\kaiou\Downloads\A3-Código Completo (100% Funcional)\cadastrados"
        
        self.conhecidos_encodings = []
        self.conhecidos_ras = []
        self.frame_atual = None
        self.face_locations = []
        self.face_names = []
        self.face_colors = []
        self.running = True
        self.ready_to_process = True 
        
        self.modo_input = None # 'cadastro' ou 'deletar'
        self.nome_digitado = ""
        self.mensagem_feedback = ""
        self.tempo_feedback = 0

        if not os.path.exists(self.diretorio_fotos):
            os.makedirs(self.diretorio_fotos, exist_ok=True)

    def treinar_sistema(self):
        novos_encodings, novos_ras = [], []
        if not os.path.exists(self.diretorio_fotos): return
        arquivos = [f for f in os.listdir(self.diretorio_fotos) if f.lower().endswith((".jpg", ".png", ".jpeg"))]
        for arquivo in arquivos:
            caminho = os.path.join(self.diretorio_fotos, arquivo)
            try:
                img = face_recognition.load_image_file(caminho)
                encodings = face_recognition.face_encodings(img)
                if len(encodings) > 0:
                    novos_encodings.append(encodings[0])
                    novos_ras.append(os.path.splitext(arquivo)[0])
            except: continue
        self.conhecidos_encodings = novos_encodings
        self.conhecidos_ras = novos_ras

    def salvar_foto_robusto(self, caminho, frame):
        try:
            is_success, im_buf_arr = cv2.imencode(".jpg", frame)
            if is_success:
                im_buf_arr.tofile(caminho)
                return True
            return False
        except: return False

    def exibir_aviso(self, texto):
        self.mensagem_feedback = texto
        self.tempo_feedback = time.time() + 3

    def _worker_reconhecimento(self):
        while self.running:
            if self.frame_atual is not None and self.ready_to_process:
                self.ready_to_process = False 
                try:
                    rgb_frame = self.frame_atual.copy()
                    locations = face_recognition.face_locations(rgb_frame, model="hog")
                    encodings = face_recognition.face_encodings(rgb_frame, locations)
                    
                    temp_names, temp_colors = [], []
                    for face_encoding in encodings:
                        ra_exibido, cor = "DESCONHECIDO", (0, 0, 255)
                        if len(self.conhecidos_encodings) > 0:
                            matches = face_recognition.compare_faces(self.conhecidos_encodings, face_encoding, tolerance=0.45)
                            dist = face_recognition.face_distance(self.conhecidos_encodings, face_encoding)
                            if len(dist) > 0 and matches[np.argmin(dist)]:
                                ra_exibido = self.conhecidos_ras[np.argmin(dist)]
                                cor = (0, 255, 0)
                        temp_names.append(ra_exibido)
                        temp_colors.append(cor)
                    self.face_locations, self.face_names, self.face_colors = locations, temp_names, temp_colors
                except: pass
                self.ready_to_process = True 
            time.sleep(0.01)

    def desenhar_guia_bancaria(self, frame, modo):
        height, width = frame.shape[:2]
        centro = (width // 2, height // 2)
        raio_x, raio_y = 150, 200
        overlay = frame.copy()
        cv2.rectangle(overlay, (0, 0), (width, height), (0, 0, 0), -1)
        cv2.ellipse(overlay, centro, (raio_x, raio_y), 0, 0, 360, (255, 255, 255), -1)
        mask = np.zeros((height, width), dtype=np.uint8)
        cv2.ellipse(mask, centro, (raio_x, raio_y), 0, 0, 360, 255, -1)
        mask_inv = cv2.bitwise_not(mask)
        img_background = cv2.bitwise_and(overlay, overlay, mask=mask_inv)
        img_foreground = cv2.bitwise_and(frame, frame, mask=mask)
        res = cv2.addWeighted(img_background, 0.5, img_foreground, 1.0, 0)
        
        if modo == 'cadastro':
            cor_borda = (0, 255, 0) if len(self.face_locations) > 0 else (0, 0, 255)
        else:
            cor_borda = (0, 165, 255) # Laranja para deletar
            
        cv2.ellipse(res, centro, (raio_x, raio_y), 0, 0, 360, cor_borda, 3)
        return res

    def iniciar_reconhecimento(self):
        video_capture = cv2.VideoCapture(0)
        nome_janela = 'Ulife FaceID Pro'
        
        # --- CONFIGURAÇÃO DE TELA CHEIA ---
        cv2.namedWindow(nome_janela, cv2.WND_PROP_FULLSCREEN)
        cv2.setWindowProperty(nome_janela, cv2.WND_PROP_FULLSCREEN, cv2.WINDOW_FULLSCREEN)
        # ----------------------------------

        threading.Thread(target=self._worker_reconhecimento, daemon=True).start()

        while True:
            ret, frame_bruto = video_capture.read()
            if not ret: break
            frame_bruto = cv2.flip(frame_bruto, 1)
            frame_exibicao = frame_bruto.copy()

            small_frame = cv2.resize(frame_bruto, (0, 0), fx=0.25, fy=0.25)
            self.frame_atual = cv2.cvtColor(small_frame, cv2.COLOR_BGR2RGB)

            if self.modo_input:
                frame_exibicao = self.desenhar_guia_bancaria(frame_bruto, self.modo_input)
                titulo = "CADASTRAR" if self.modo_input == 'cadastro' else "DELETAR"
                cv2.putText(frame_exibicao, f"MODO: {titulo}", (30, 40), cv2.FONT_HERSHEY_DUPLEX, 0.6, (255, 255, 255), 1)
                cv2.putText(frame_exibicao, f"RA: {self.nome_digitado}", (30, 70), cv2.FONT_HERSHEY_DUPLEX, 0.8, (0, 255, 255), 1)
            else:
                for (top, right, bottom, left), name, cor in zip(self.face_locations, self.face_names, self.face_colors):
                    top *= 4; right *= 4; bottom *= 4; left *= 4
                    cv2.rectangle(frame_exibicao, (left, top), (right, bottom), cor, 2)
                    cv2.putText(frame_exibicao, f"RA: {name}", (left, bottom + 25), cv2.FONT_HERSHEY_DUPLEX, 0.7, cor, 1)
                
                if time.time() < self.tempo_feedback:
                    cv2.putText(frame_exibicao, self.mensagem_feedback, (50, 240), cv2.FONT_HERSHEY_DUPLEX, 0.8, (0, 255, 255), 2)

            cv2.imshow(nome_janela, frame_exibicao)
            key = cv2.waitKey(1)

            if self.modo_input is None:
                if key == ord('q'): break
                elif key == ord('c'): self.modo_input = 'cadastro'; self.nome_digitado = ""
                elif key == ord('d'): self.modo_input = 'deletar'; self.nome_digitado = ""
            
            else:
                if key == 27: # ESC
                    self.modo_input = None
                elif key == 13: # ENTER
                    nome_alvo = self.nome_digitado.strip()
                    caminho_alvo = os.path.join(self.diretorio_fotos, f"{nome_alvo}.jpg")
                    
                    if not nome_alvo:
                        self.exibir_aviso("ERRO: DIGITE O RA!")
                    
                    elif self.modo_input == 'cadastro':
                        if len(self.face_locations) == 0:
                            self.exibir_aviso("FALHA: ROSTO NAO POSICIONADO!")
                        else:
                            if self.salvar_foto_robusto(caminho_alvo, frame_bruto):
                                self.exibir_aviso("SUCESSO: CADASTRADO!")
                                self.treinar_sistema()
                                self.modo_input = None
                            else:
                                self.exibir_aviso("ERRO AO GRAVAR ARQUIVO!")

                    elif self.modo_input == 'deletar':
                        if os.path.exists(caminho_alvo):
                            try:
                                os.remove(caminho_alvo)
                                self.exibir_aviso("SUCESSO: RA DELETADO!")
                                self.treinar_sistema()
                                self.modo_input = None
                            except:
                                self.exibir_aviso("ERRO AO DELETAR!")
                        else:
                            self.exibir_aviso("RA NAO ENCONTRADO!")
                
                elif key == 8: # BACKSPACE
                    self.nome_digitado = self.nome_digitado[:-1]
                elif 32 <= key <= 126: # DIGITAÇÃO
                    self.nome_digitado += chr(key)

        video_capture.release()
        cv2.destroyAllWindows()

if __name__ == "__main__":
    app = SistemaAcessoUlife()
    app.treinar_sistema()
    app.iniciar_reconhecimento()
